cmake_minimum_required(VERSION 3.20)
project(DirectorCut VERSION 0.3.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(DIRECTORCUT_BUILD_TESTS "Build tests" ON)
option(DIRECTORCUT_BUILD_QT_UI "Build Qt/QML desktop shell when Qt 6 is available" ON)
option(DIRECTORCUT_BUILD_GSTREAMER "Build native GStreamer playback backend when development packages are available" ON)

find_package(SQLite3 REQUIRED)

add_library(directorcut_core
    core/src/timeline.cpp
    core/src/commands.cpp
    core/src/event_store.cpp
    core/src/director.cpp
    core/src/transcript.cpp
    core/src/media.cpp
)
target_include_directories(directorcut_core PUBLIC core/include)
target_link_libraries(directorcut_core PUBLIC SQLite::SQLite3)
if (MSVC)
  target_compile_options(directorcut_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(directorcut_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(directorcut apps/cli/main.cpp)
target_link_libraries(directorcut PRIVATE directorcut_core)

if (DIRECTORCUT_BUILD_GSTREAMER)
  find_package(PkgConfig QUIET)
  if (PkgConfig_FOUND)
    pkg_check_modules(GSTREAMER QUIET IMPORTED_TARGET gstreamer-1.0>=1.22)
    pkg_check_modules(GES QUIET IMPORTED_TARGET gst-editing-services-1.0>=1.22)
  endif()
  if (GSTREAMER_FOUND)
    add_library(directorcut_gstreamer core/src/gstreamer_playback.cpp)
    target_include_directories(directorcut_gstreamer PUBLIC core/include)
    target_link_libraries(directorcut_gstreamer PUBLIC directorcut_core PkgConfig::GSTREAMER)
    target_compile_definitions(directorcut_gstreamer PUBLIC DIRECTORCUT_HAS_GSTREAMER=1)
    if (GES_FOUND)
      target_link_libraries(directorcut_gstreamer PUBLIC PkgConfig::GES)
      target_compile_definitions(directorcut_gstreamer PUBLIC DIRECTORCUT_HAS_GES=1)
      message(STATUS "GStreamer + Editing Services detected: native playback backend enabled")
    else()
      message(STATUS "GStreamer detected without Editing Services: playback backend enabled, GES timeline integration disabled")
    endif()
  else()
    message(STATUS "GStreamer development packages not found: native playback backend skipped")
  endif()
endif()

if (DIRECTORCUT_BUILD_TESTS)
  enable_testing()
  add_executable(directorcut_tests tests/test_core.cpp)
  target_link_libraries(directorcut_tests PRIVATE directorcut_core)
  add_test(NAME directorcut_core_tests COMMAND directorcut_tests)
  if (TARGET directorcut_gstreamer)
    add_executable(directorcut_gstreamer_tests tests/test_gstreamer.cpp)
    target_link_libraries(directorcut_gstreamer_tests PRIVATE directorcut_gstreamer)
    add_test(NAME directorcut_gstreamer_tests COMMAND directorcut_gstreamer_tests)
  endif()
endif()

if (DIRECTORCUT_BUILD_QT_UI)
  find_package(Qt6 QUIET COMPONENTS Quick Qml QuickControls2)
  if (Qt6_FOUND)
    qt_add_executable(directorcut_desktop native-ui/main.cpp)
    qt_add_qml_module(directorcut_desktop URI DirectorCut VERSION 1.0 QML_FILES native-ui/Main.qml)
    target_link_libraries(directorcut_desktop PRIVATE Qt6::Quick Qt6::Qml Qt6::QuickControls2 directorcut_core)
    if (TARGET directorcut_gstreamer)
      target_link_libraries(directorcut_desktop PRIVATE directorcut_gstreamer)
      target_compile_definitions(directorcut_desktop PRIVATE DIRECTORCUT_NATIVE_GSTREAMER=1)
    endif()
  else()
    message(STATUS "Qt 6 not found: building core + CLI only. QML desktop source remains in native-ui/.")
  endif()
endif()
